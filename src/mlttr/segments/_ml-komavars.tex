% Some new KOMAvars.

\newkomavar{fromlastname}
\newkomavar{fromstreet}
\newkomavar{fromzip}
\newkomavar{fromlinkedin}

% An "immutable" komavar.
\setkomavar{backaddressseparator}{\, \ensuremath{\bullet} \,}

% Empty the signature variable
% (anyway, should not be used when inserting signatures as graphics).
\setkomavar{signature}{}

% Assert that a minimal set of komavars is defined.
\newcommand*{\Providekomavar}[2]{%
  \Ifkomavarempty{#1}{\setkomavar{#1}{#2}}{}
}

% Checks all komavars at the begin of the document
\AtBeginDocument{%

  % Assigns a dummy value to every unset komavar
  \Providekomavar{fromlastname}{<lastname>}
  \Providekomavar{fromgivenname}{<givenname>}
  \Providekomavar{fromstreet}{<street>}
  \Providekomavar{fromzip}{<zip>}
  \Providekomavar{fromemail}{<email>}
  \Providekomavar{frommobilephone}{<mobilephone>}
  \Providekomavar{birthday}{<birthday>}
  \Providekomavar{birthplace}{<birthplace>}
  \Providekomavar{iban}{<iban>}
  \Providekomavar{bic}{<bic>}
  \Providekomavar{bank}{<bank>}
  % \Providekomavar{signature}{snoke_signature.jpg}
  % \Providekomavar{konterfei}{snoke_konterfei_square.jpg}
  \Providekomavar{place}{<place>}
  \Providekomavar{dateyear}{\the\year}
  \Providekomavar{datemonth}{\the\month}
  \Providekomavar{dateday}{\the\day}
  \Providekomavar{subject}{<subject>}
  \Providekomavar{yourref}{<aktenzeichen>}
  \Providekomavar{description}{<description>}
  \Providekomavar{keywords}{<keywords>}

  % Assigns values to the composite komavars
  % This must be so cumbersome due to the
  % non-full expandibility of `\usekomavar`
  % if one wants to use the composite ones
  % as the usual ones.

  \usekomavar[\edef\KOMAfromgivenname]{fromgivenname}%
  \usekomavar[\edef\KOMAfromlastname]{fromlastname}%
  \setkomavar{fromname}{%
    \KOMAfromgivenname{} \KOMAfromlastname{}
  }

  \usekomavar[\edef\KOMAfromstreet]{fromstreet}%
  \usekomavar[\edef\KOMAfromzip]{fromzip}%
  \setkomavar{fromaddress}{%
    \KOMAfromstreet{} \KOMAfromzip{}
  }

  \def\edef@two@digits#1#2{\edef#1{\two@digits{#2}}}
  \usekomavar[\edef@two@digits\KOMAdateyear]{dateyear}%
  \usekomavar[\edef@two@digits\KOMAdatemonth]{datemonth}%
  \usekomavar[\edef@two@digits\KOMAdateday]{dateday}%
  \setkomavar{date}{%
    \KOMAdateyear{}-\KOMAdatemonth{}-\KOMAdateday{}%
  }

  \usekomavar[\edef\KOMAiban]{iban}%
  \usekomavar[\edef\KOMAbic]{bic}%
  \usekomavar[\edef\KOMAbank]{bank}%
  \setkomavar{bankdata}{%
    \KOMAiban{}, \KOMAbic{}, \KOMAbank{}
  }
  % actually, bic and bank could be derived from the iban
  % but does there exist an appropriate package -:)?

}
